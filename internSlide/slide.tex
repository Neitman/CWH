\section{PHẦN 1: GIỚI THIỆU}

\begin{frame}{1.1. Công việc được giao}
    
    
    \vspace{0.3cm}
    \begin{block}{Công việc}
        \begin{itemize}
            \item Nghiên cứu \& phát triển ứng dụng Web thời gian thực đồng bộ phát video nhóm qua giao thức WebSocket.
            \item Xây dựng hệ thống xác thực an toàn (JWT, OTP Email, Bcrypt, Token kép Access/Refresh Token).
            \item Quản lý phòng xem, danh sách phát cá nhân, tải lên \& stream video local tùy chỉnh, trò chuyện Timestamp, phân quyền chủ phòng và Dashboard quản trị.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{1.2. Giới thiệu công nghệ WebSocket \& Socket.IO}
    \begin{block}{Giao thức WebSocket (RFC 6455)}
        \begin{itemize}
            \item Kênh truyền thông \textbf{song công toàn phần (Full-duplex)} trên 1 kết nối TCP duy nhất.
            \item Duy trì kết nối mở liên tục: Client và Server chủ động Push dữ liệu tức thì.
            \item \textbf{Tối ưu hiệu năng:} Frame nhị phân rất nhỏ ($2B - 10B$), loại bỏ HTTP Header overhead, tối ưu dung lượng truyền tải.
        \end{itemize}
    \end{block}
    
    \vspace{0.1cm}
    \begin{block}{Thư viện Socket.IO (Lớp trừu tượng trên nền WebSocket)}
        \begin{itemize}
            \item \textbf{Cơ chế Room xem nhóm:} Quản lý nhóm Socket để phát thông điệp đồng bộ (broadcast) tức thời tới toàn bộ thành viên trong phòng.
            \item \textbf{Độ tin cậy cao:} Tự động kết nối lại (Auto-reconnect) và tự động Fallback sang HTTP long-polling khi mạng kém.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{1.3. Khảo sát Bảo mật \& Kiến trúc Cơ sở dữ liệu}
    \begin{block}{Xác thực \& Bảo mật (JWT + Bcrypt)}
        \begin{itemize}
            \item \textbf{JWT (JSON Web Token):} Cơ chế phi trạng thái (Stateless), mã hóa chữ ký số HMAC-SHA256. Đính kèm dễ dàng trong REST API Header và quá trình khởi tạo kết nối WebSocket (Handshake).
            \item \textbf{Bcrypt:} Thuật toán băm mật khẩu 1 chiều đính kèm Salt ngẫu nhiên chống vét cạn.
        \end{itemize}
    \end{block}
    
    \vspace{0.2cm}
    \begin{block}{Kiến trúc Cơ sở dữ liệu}
        \begin{itemize}
            \item \textbf{PostgreSQL (RDBMS):} Lưu trữ bền vững tài khoản người dùng, thông tin phòng xem, danh sách phát cá nhân và lịch sử tin nhắn trò chuyện.
            \item \textbf{Redis (In-memory Cache):} Lưu trữ trạng thái phát thời gian thực (tiến độ video, trạng thái phát/tạm dừng, thời gian cập nhật), mã OTP, Refresh Token với tốc độ phản hồi $<1ms$.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{2.2. Module Xác thực OTP Email \& Cơ chế Token Kép}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Xác thực OTP Email (Nodemailer)}
                \begin{itemize}
                    \item Gửi mã xác thực 6 chữ số qua SMTP Email cho Đăng ký \& Quên mật khẩu.
                    \item Lưu mã OTP đệm trong Redis với thời gian hết hạn (TTL) là $5$ phút.
                    \item Tự động gửi Email chào mừng khi xác minh thành công.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Cơ chế Token Kép \& Bảo mật}
                \begin{itemize}
                    \item \textbf{Access Token ($5$ phút):} Xác thực REST API \& WebSocket Handshake.
                    \item \textbf{Refresh Token ($7$ ngày):} Lưu an toàn trong HTTP-Only Cookie \& bộ đệm Redis.
                    \item Tự động gia hạn Access Token khi hết hạn và chủ động thu hồi token khi đăng xuất.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.3. Triển khai Đồng bộ Trạng thái Video}
    \begin{block}{Khắc phục vòng lặp sự kiện (Event Loop Feedback Loop)}
        Khi Client nhận lệnh đồng bộ từ Server và gọi API trình phát (thay đổi vị trí, tạm dừng), sự kiện thay đổi trạng thái trình phát tự động kích hoạt làm gửi ngược tin nhắn lên Server. Hệ thống sử dụng \textbf{cờ tạm khóa trạng thái đồng bộ} để ngăn ngừa phản hồi trùng lặp.
    \end{block}
    
    \vspace{0.2cm}
    \begin{block}{Tính toán mốc thời gian phát (Progress Calculation)}
        Khi người dùng mới vào phòng, Server tự động tính thời gian video đã trôi qua dựa trên mốc thời gian cập nhật gần nhất lưu trong Redis:
        \[ \text{Tiến độ hiện tại (s)} = \text{Tiến độ ban đầu (s)} + \frac{\text{Thời gian hiện tại (ms)} - \text{Thời gian cập nhật gần nhất (ms)}}{1000} \]
        Đảm bảo các Client mới gia nhập phòng luôn cập nhật đúng mốc thời gian hiện tại.
    \end{block}
\end{frame}

\begin{frame}{2.4. Module Tải lên \& Stream Video Tùy chỉnh (Custom Video)}
    \begin{itemize}
        \item \textbf{Tải lên tệp Video địa phương (Multer Upload):}
        \begin{itemize}
            \item Cho phép tải lên các tệp video local (\texttt{.mp4}, \texttt{.webm}, \texttt{.mkv}, \texttt{.avi}, \texttt{.mov}) với dung lượng lên đến $10GB$.
            \item Lưu trữ an toàn tại thư mục lưu trữ tĩnh của máy chủ.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Truyền dữ liệu phát phân đoạn chuẩn HTTP Range Requests (Code 206):}
        \begin{itemize}
            \item Xử lý yêu cầu truyền dữ liệu phân đoạn từ phía Client qua HTTP Header (Range Requests).
            \item Server trả về phản hồi dữ liệu phân đoạn (HTTP 206 Partial Content) qua cơ chế Stream I/O.
            \item Giúp người dùng tua mượt mà không cần nạp trước toàn bộ tệp lớn về máy.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Cơ chế dọn dẹp bộ nhớ đĩa tự động (Auto Disk Cleanup):}
        \begin{itemize}
            \item Tự động xóa tệp video khỏi đĩa cứng ngay khi gỡ video khỏi hàng chờ phát.
            \item Cron Job ngầm định kỳ mỗi $1$ giờ tự động xóa các tệp video mồ côi quá $6$ giờ.
        \end{itemize}
    \end{itemize}
\end{frame}

                    \item Kỹ năng thiết kế hệ thống phân quyền, xử lý stream nhị phân và dọn dẹp bộ nhớ.
