\section{PHẦN 1: GIỚI THIỆU}

\begin{frame}{1.1. Phát biểu bài toán \& Công việc được giao}
    \begin{block}{Đặt vấn đề thực tế}
        \begin{itemize}
            \item Nhu cầu xem video nhóm từ xa (Co-watching) ngày càng gia tăng.
            \item Các giải pháp thủ công (đếm ngược 3-2-1) bộc lộ hạn chế lớn: lệch khung hình do trễ mạng ($2s - 10s$) và mất tính tương tác khi tạm dừng video.
        \end{itemize}
    \end{block}
    
    \vspace{0.3cm}
    \begin{block}{Công việc được giao}
        \begin{itemize}
            \item Nghiên cứu \& phát triển ứng dụng Web thời gian thực đồng bộ trạng thái xem video nhóm với độ trễ siêu thấp ($<1s$).
            \item Triển khai xác thực bảo mật JWT \& mã hóa mật khẩu Bcrypt.
            \item Quản lý phòng xem, danh sách hàng đợi (Queue) và trò chuyện (Chat) đính kèm mốc thời gian (Timestamp).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{1.2. Khảo sát công nghệ truyền thông thời gian thực}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{HTTP Polling (Truyền thống)}
                \begin{itemize}
                    \item Client gửi HTTP Request liên tục.
                    \item \textbf{Nhược điểm:} Header overhead lớn ($500B - 2KB$/req), gây quá tải CPU/RAM máy chủ.
                    \item \textbf{Độ trễ cao:} $1s - 5s$, không thể chủ động Push dữ liệu từ Server.
                \end{itemize}
            \end{alertblock}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{WebSocket \& Socket.IO (Lựa chọn)}
                \begin{itemize}
                    \item Kênh truyền song công (Full-duplex) trên 1 kết nối TCP duy nhất.
                    \item \textbf{Ưu điểm:} Overhead cực nhỏ ($2B - 10B$), độ trễ siêu thấp ($<100ms$).
                    \item Server chủ động Push dữ liệu; Socket.IO hỗ trợ auto-reconnect \& quản lý Room.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{1.3. Khảo sát Bảo mật \& Kiến trúc Cơ sở dữ liệu}
    \begin{block}{Xác thực \& Bảo mật (JWT + Bcrypt)}
        \begin{itemize}
            \item \textbf{JWT (JSON Web Token):} Phi trạng thái (Stateless), chữ ký số HMAC-SHA256. Dễ dàng đính kèm trong REST Header và WebSocket Handshake (`socket.handshake.auth.token`).
            \item \textbf{Bcrypt:} Hàm băm mật khẩu 1 chiều tích hợp Salt ngẫu nhiên chống vét cạn.
        \end{itemize}
    \end{block}
    
    \vspace{0.2cm}
    \begin{block}{Kiến trúc Cơ sở dữ liệu Lai (Hybrid Database)}
        \begin{itemize}
            \item \textbf{PostgreSQL (RDBMS):} Lưu trữ bền vững tài khoản người dùng, thông tin phòng xem, lịch sử tin nhắn trò chuyện.
            \item \textbf{Redis (In-memory Cache):} Lưu trữ trạng thái phát thời gian thực (`progress`, `isPlaying`, `lastUpdated`) với tốc độ phản hồi $<1ms$.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{2.2. Triển khai Module Xác thực \& Quản lý phòng xem}
    \begin{itemize}
        \item \textbf{REST API Authentication:}
        \begin{itemize}
            \item Route Đăng ký, Đăng nhập trả về Token JWT mã hóa.
            \item Middleware `authenticateToken` giải mã Header `Authorization: Bearer <token>`.
        \end{itemize}
        \vspace{0.3cm}
        \item \textbf{WebSocket Handshake \& Room Management:}
        \begin{itemize}
            \item Kiểm tra Token JWT ngay tại thời điểm bắt tay Handshake kết nối Socket.
            \item Nhóm các kết nối Socket vào từng phòng xem riêng biệt qua lệnh `socket.join(roomId)`.
            \item Tự động đồng bộ và phát danh sách thành viên đang online trong phòng.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{2.3. Triển khai Đồng bộ Video \& Thuật toán Khử trễ}
    \begin{block}{Khắc phục vòng lặp sự kiện (Event Loop Feedback Loop)}
        Client nhận lệnh đồng bộ từ Server và gọi API trình phát (`seekTo`, `pauseVideo`) sẽ làm kích hoạt lại sự kiện `onStateChange`. Hệ thống sử dụng \textbf{cờ `isSyncing = true`} để tạm dừng phát sự kiện ngược lên Server trong $300ms$.
    \end{block}
    
    \vspace{0.2cm}
    \begin{block}{Thuật toán bù sai lệch thời gian phát (Drift Compensation)}
        Khi người dùng mới vào phòng, Server tự động tính thời gian video đã trôi qua dựa trên mốc thời gian lưu trong Redis:
        \[ \text{progress} = \text{progress\_gốc} + \frac{\text{Date.now()} - \text{lastUpdated}}{1000} \]
        Đảm bảo mọi thành viên trong phòng luôn xem đúng từng khung hình.
    \end{block}
\end{frame}

\begin{frame}{2.4. Triển khai Module Trò chuyện nhóm (Chat Timestamp)}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{\textcolor{primary}{Gửi tin nhắn đính kèm Timestamp}}
            \begin{itemize}
                \item Server tự động trích xuất `videoTimestamp` hiện tại của video khi Client gửi tin nhắn.
                \item Lưu bản ghi vào PostgreSQL và broadcast tới tất cả thành viên trong phòng.
            \end{itemize}
        \end{column}
        
        \begin{column}{0.5\textwidth}
            \textbf{\textcolor{primary}{Tương tác nhảy mốc thời gian}}
            \begin{itemize}
                \item Tin nhắn hiển thị kèm mốc thời gian (ví dụ \texttt{02:45}).
                \item Thành viên nhấp vào mốc thời gian $\rightarrow$ Trình phát video lập tức nhảy tới khoảnh khắc đó.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

    \begin{center}
        \textbf{\Large Giao diện Ứng dụng Web Single Page Application (SPA)}
    \end{center}
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Màn hình Đăng nhập / Đăng ký:} Giao diện hiện đại, xác thực dữ liệu đầu vào và lưu trữ Token bảo mật.
        \item \textbf{Màn hình Dashboard Quản lý phòng:} Tạo phòng xem mới và truy cập nhanh bằng Mã phòng (Room ID).
        \item \textbf{Màn hình Phòng xem video trực tuyến:} Trình phát HTML5 / YouTube Player responsive, Khung danh sách phát bài hát (Queue), Khung trò chuyện nhóm đính kèm Timestamp và Danh sách thành viên online.
    \end{enumerate}
\end{frame}
