\section{PHẦN 1: GIỚI THIỆU}

\begin{frame}{1.1. Phát biểu bài toán \& Công việc được giao}
    \begin{block}{Đặt vấn đề thực tế}
        \begin{itemize}
            \item Nhu cầu xem video nhóm từ xa (Co-watching) ngày càng gia tăng mạnh mẽ.
            \item Giải pháp thủ công (đếm ngược 3-2-1) bộc lộ hạn chế lớn: lệch độ trễ mạng ($2s - 10s$) và làm mất tính tương tác thời gian thực khi có người tạm dừng.
        \end{itemize}
    \end{block}
    
    \vspace{0.3cm}
    \begin{block}{Mục tiêu \& Công việc được giao}
        \begin{itemize}
            \item Nghiên cứu \& phát triển ứng dụng Web thời gian thực đồng bộ phát video nhóm với độ trễ siêu thấp ($<1s$).
            \item Xây dựng hệ thống xác thực an toàn (JWT, OTP Email, Bcrypt, Token kép Access/Refresh Token).
            \item Quản lý phòng xem, danh sách phát cá nhân, tải lên \& stream video local tùy chỉnh, trò chuyện Timestamp, phân quyền chủ phòng và Dashboard quản trị.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{1.2. Giới thiệu công nghệ WebSocket \& Socket.IO}
    \begin{block}{Giao thức WebSocket (RFC 6455)}
        \begin{itemize}
            \item Kênh truyền thông \textbf{song công toàn phần (Full-duplex)} trên 1 kết nối TCP duy nhất.
            \item Duy trì kết nối mở liên tục: Client và Server chủ động Push dữ liệu tức thì.
            \item \textbf{Tối ưu hiệu năng:} Frame nhị phân rất nhỏ ($2B - 10B$), loại bỏ HTTP Header overhead, độ trễ siêu thấp ($<100ms$).
        \end{itemize}
    \end{block}
    
    \vspace{0.1cm}
    \begin{block}{Thư viện Socket.IO (Lớp trừu tượng trên nền WebSocket)}
        \begin{itemize}
            \item \textbf{Cơ chế Room xem nhóm:} Quản lý nhóm Socket qua lệnh \texttt{socket.join(roomId)} để broadcast sự kiện đồng bộ tức thời tới toàn phòng.
            \item \textbf{Độ tin cậy cao:} Tự động kết nối lại (Auto-reconnect) và tự động Fallback sang HTTP long-polling khi mạng kém.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{1.3. Khảo sát Bảo mật \& Kiến trúc Cơ sở dữ liệu}
    \begin{block}{Xác thực \& Bảo mật (JWT + Bcrypt)}
        \begin{itemize}
            \item \textbf{JWT (JSON Web Token):} Cơ chế phi trạng thái (Stateless), mã hóa chữ ký số HMAC-SHA256. Đính kèm dễ dàng trong REST API Header và WebSocket Handshake (\texttt{socket.handshake.auth.token}).
            \item \textbf{Bcrypt:} Thuật toán băm mật khẩu 1 chiều đính kèm Salt ngẫu nhiên chống vét cạn.
        \end{itemize}
    \end{block}
    
    \vspace{0.2cm}
    \begin{block}{Kiến trúc Cơ sở dữ liệu Lai (Hybrid Database)}
        \begin{itemize}
            \item \textbf{PostgreSQL (RDBMS):} Lưu trữ bền vững tài khoản người dùng, thông tin phòng xem, danh sách phát cá nhân và lịch sử tin nhắn trò chuyện.
            \item \textbf{Redis (In-memory Cache):} Lưu trữ trạng thái phát thời gian thực (\texttt{progress}, \texttt{isPlaying}, \texttt{lastUpdated}), OTP code, Refresh Token với tốc độ phản hồi $<1ms$.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{2.2. Module Xác thực OTP Email \& Cơ chế Token Kép}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Xác thực OTP Email (Nodemailer)}
                \begin{itemize}
                    \item Gửi mã xác thực 6 chữ số qua SMTP Email cho Đăng ký \& Quên mật khẩu.
                    \item Lưu mã OTP đệm trong Redis với thời gian hết hạn (TTL) là $5$ phút (\texttt{300s}).
                    \item Tự động gửi Email chào mừng khi xác minh thành công.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Cơ chế Token Kép \& Bảo mật}
                \begin{itemize}
                    \item \textbf{Access Token ($5$ phút):} Xác thực REST API \& WebSocket Handshake.
                    \item \textbf{Refresh Token ($7$ ngày):} Lưu an toàn trong HTTP-Only Cookie \& Redis (\texttt{wp:refreshtoken:<id>}).
                    \item Route \texttt{/refresh} gia hạn tự động; \texttt{/logout} chủ động thu hồi token.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.3. Triển khai Đồng bộ Video \& Thuật toán Khử trễ}
    \begin{block}{Khắc phục vòng lặp sự kiện (Event Loop Feedback Loop)}
        Khi Client nhận lệnh đồng bộ từ Server và gọi API trình phát (\texttt{seekTo}, \texttt{pauseVideo}), sự kiện \texttt{onStateChange} tự động bật làm gửi ngược tin nhắn lên Server. Hệ thống sử dụng \textbf{cờ \texttt{isSyncing = true}} để tạm khóa gửi tin nhắn trong $300ms$.
    \end{block}
    
    \vspace{0.2cm}
    \begin{block}{Thuật toán bù sai lệch thời gian phát (Drift Compensation)}
        Khi người dùng mới vào phòng, Server tự động tính thời gian video đã trôi qua dựa trên mốc \texttt{lastUpdated} lưu trong Redis:
        \[ \text{progress (s)} = \text{progress\_gốc (s)} + \frac{\text{Thời gian hiện tại (ms)} - \text{lastUpdated (ms)}}{1000} \]
        Đảm bảo các Client luôn xem đúng mốc thời gian với sai lệch $<300ms$.
    \end{block}
\end{frame}

\begin{frame}{2.4. Module Tải lên \& Stream Video Tùy chỉnh (Custom Video)}
    \begin{itemize}
        \item \textbf{Tải lên tệp Video địa phương (Multer Upload):}
        \begin{itemize}
            \item Cho phép tải lên các tệp video local (\texttt{.mp4}, \texttt{.webm}, \texttt{.mkv}, \texttt{.avi}, \texttt{.mov}) với dung lượng lên đến $10GB$.
            \item Lưu trữ an toàn tại thư mục tĩnh \texttt{public/uploads/videos}.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Truyền dữ liệu phát phân đoạn chuẩn HTTP Range Requests (Code 206):}
        \begin{itemize}
            \item Route \texttt{GET /api/videos/stream/:filename} xử lý header \texttt{Range: bytes=start-end}.
            \item Server trả về phản hồi \texttt{206 Partial Content} qua Stream I/O Node.js (\texttt{fs.createReadStream}).
            \item Giúp người dùng tua (seek) mượt mà không cần nạp trước toàn bộ tệp lớn về máy.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Cơ chế dọn dẹp bộ nhớ đĩa tự động (Auto Disk Cleanup):}
        \begin{itemize}
            \item Hàm \texttt{cleanupCustomVideoFile()} tự động xóa tệp bằng \texttt{fs.unlinkSync()} khi gỡ video khỏi Queue.
            \item Cron Job ngầm định kỳ mỗi $1$ giờ tự động xóa các tệp video mồ côi quá $6$ giờ.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{3.3. Phân tích 4 lỗi kỹ thuật đã xử lý triệt để}
    \begin{enumerate}[leftmargin=1.2cm]
        \item \textbf{Lỗi vòng lặp sự kiện vô hạn (Event Loop Feedback):} Trình phát gọi API kích hoạt lại \texttt{onStateChange}. \textit{Giải pháp:} Sử dụng cờ cản \texttt{isSyncing} tạm khóa phát tin nhắn ngược lên Server trong $300ms$.
        \vspace{0.15cm}
        \item \textbf{Lỗi trôi thời gian phát (Time Drift):} Người vào sau bị lệch pha. \textit{Giải pháp:} Thuật toán Drift Compensation tự động tính toán thời gian trôi qua dựa trên \texttt{lastUpdated} trong Redis.
        \vspace{0.15cm}
        \item \textbf{Lỗi rò rỉ tệp tin rác trên đĩa:} Video tự tải lên không bị xóa khi gỡ bài. \textit{Giải pháp:} Xây dựng hàm \texttt{cleanupCustomVideoFile()} xóa tệp tin bằng \texttt{fs.unlinkSync()} ngay khi bỏ bài khỏi Queue.
        \vspace{0.15cm}
        \item \textbf{Lỗi Token JWT hết hạn:} Ngắt kết nối giữa chừng. \textit{Giải pháp:} Đăng ký handler \texttt{auth-error} hiển thị thông báo và yêu cầu chuyển hướng đăng nhập/refresh token.
    \end{enumerate}
\end{frame}
